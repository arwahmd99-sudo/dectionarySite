إعدادات Google Sheets والرمز السري

ربط الموقع بملف Google Sheets سحابياً وتغيير الرمز السري للمعلمة.

ربط ملف Google Sheets

1. أنشئي جدول بيانات جديد في Google Sheets.
2. من القائمة اختاري Extensions > Apps Script والصقي كود البرمجة (Apps Script) المذكور في الأسفل.
3. اضغطي على Deploy > New deployment (اختر Web app، و Execute as: Me، و Who has access: Anyone)، ثم انسخي رابط الويب (Web App URL) والصقيه هنا:

كود Google Apps Script (يُنسخ ويلصق في ملف الشيت): 

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var data = [];
  // Skip header row if exists
  for (var i = 1; i < rows.length; i++) {
    if (rows[i][0]) {
      data.push({
        id: rows[i][0],
        name: rows[i][1],
        className: rows[i][2],
        createdAt: Number(rows[i][3]) || Date.now(),
        scores: {
          w1: rows[i][4] === true || rows[i][4] === "true",
          w2: rows[i][5] === true || rows[i][5] === "true",
          w3: rows[i][6] === true || rows[i][6] === "true",
          w4: rows[i][7] === true || rows[i][7] === "true",
          w5: rows[i][8] === true || rows[i][8] === "true",
          w6: rows[i][9] === true || rows[i][9] === "true",
          w7: rows[i][10] === true || rows[i][10] === "true",
          w8: rows[i][11] === true || rows[i][11] === "true"
        }
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var payload = JSON.parse(e.postData.contents);
    
    // Clear and write header
    sheet.clear();
    sheet.appendRow(["ID", "Name", "ClassName", "CreatedAt", "w1", "w2", "w3", "w4", "w5", "w6", "w7", "w8"]);
    
    var students = payload.students || [];
    for (var i = 0; i < students.length; i++) {
      var s = students[i];
      var sc = s.scores || {};
      sheet.appendRow([
        s.id,
        s.name,
        s.className,
        s.createdAt,
        !!sc.w1, !!sc.w2, !!sc.w3, !!sc.w4, !!sc.w5, !!sc.w6, !!sc.w7, !!sc.w8
      ]);
    }
    return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
